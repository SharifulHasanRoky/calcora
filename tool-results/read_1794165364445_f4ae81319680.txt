    98→/** PMT — standard amortizing payment. r = periodic rate, n = total periods. */
    99→function pmt(principal: number, r: number, n: number): number {
   100→  if (n <= 0) return 0;
   101→  if (r === 0) return principal / n;
   102→  return (principal * r * Math.pow(1 + r, n)) / (Math.pow(1 + r, n) - 1);
   103→}
   104→
   105→/** Future value of a present sum with periodic compounding. */
   106→function fvLumpSum(pv: number, r: number, n: number, t: number): number {
   107→  if (n <= 0) return pv;
   108→  return pv * Math.pow(1 + r / n, n * t);
   109→}
   110→
   111→/** Future value of an ordinary annuity (end-of-period contributions). */
   112→function fvAnnuity(pmt: number, r: number, n: number, t: number): number {
   113→  if (n <= 0) return 0;
   114→  const periods = n * t;
   115→  const rate = r / n;
   116→  if (rate === 0) return pmt * periods;
   117→  return pmt * ((Math.pow(1 + rate, periods) - 1) / rate);
   118→}

   124→function LoanCalc() {
   125→  const [currency, setCurrency] = React.useState<string>("USD");
   126→  const [principal, setPrincipal] = React.useState("250000");
   127→  const [rate, setRate] = React.useState("6.5");
   128→  const [term, setTerm] = React.useState("30");
   129→
   130→  const calc = React.useMemo(() => {
   131→    const P = parseNumber(principal);
   132→    const annualRate = parseNumber(rate) / 100;
   133→    const years = parseNumber(term);
   134→    const n = Math.max(1, Math.round(years * 12));
   135→    const r = annualRate / 12;
   136→    const monthly = pmt(P, r, n);
   137→    const totalPaid = monthly * n;
   138→    const totalInterest = totalPaid - P;
   139→
   140→    const rows: { month: number; payment: number; principal: number; interest: number; balance: number }[] = [];
   141→    let balance = P;
   142→    const limit = Math.min(12, n);
   143→    for (let i = 1; i <= limit; i++) {
   144→      const interestPart = balance * r;
   145→      const principalPart = monthly - interestPart;
   146→      balance -= principalPart;
   147→      rows.push({
   148→        month: i,
   149→        payment: monthly,
   150→        principal: principalPart,
   151→        interest: interestPart,
   152→        balance: Math.max(0, balance),
   153→      });
   154→    }
   155→    return { P, monthly, totalPaid, totalInterest, n, rows };
   156→  }, [principal, rate, term]);

   158→  return (

   173→          {calc.n} monthly payments
   174→        </div>
   175→      </div>
   176→
   177→      <div className="grid sm:grid-cols-3 gap-4">
   178→        <CopyableStat label="Monthly Payment" displayValue={money(calc.monthly, currency)} accent="primary" />
   179→        <ResultStat label="Total Paid" value={money(calc.totalPaid, currency)} accent="default" />

   182→
   183→      <div className="space-y-3">
   184→        <div className="flex items-center justify-between">
   185→          <h3 className="text-sm font-semibold">Amortization preview (first 12 months)</h3>
   186→          <CopyRowButton

   239→function MortgageCalc() {
   240→  const [currency, setCurrency] = React.useState<string>("USD");
   241→  const [homePrice, setHomePrice] = React.useState("400000");
   242→  const [downPayment, setDownPayment] = React.useState("80000");
   243→  const [rate, setRate] = React.useState("6.8");
   244→  const [term, setTerm] = React.useState("30");
   245→  const [propertyTaxRate, setPropertyTaxRate] = React.useState("1.1");
   246→  const [homeInsurance, setHomeInsurance] = React.useState("1400");
   247→  const [pmiRate, setPmiRate] = React.useState("0.5");
   248→
   249→  const calc = React.useMemo(() => {
   250→    const price = parseNumber(homePrice);
   251→    const down = parseNumber(downPayment);
   252→    const principal = Math.max(0, price - down);
   253→    const annualRate = parseNumber(rate) / 100;
   254→    const years = parseNumber(term);
   255→    const n = Math.max(1, Math.round(years * 12));
   256→    const r = annualRate / 12;
   257→
   258→    const pi = pmt(principal, r, n); // principal + interest
   259→    const monthlyPropTax = (parseNumber(propertyTaxRate) / 100 * price) / 12;
   260→    const monthlyIns = parseNumber(homeInsurance) / 12;

   264→
   265→    const totalMonthly = pi + monthlyPropTax + monthlyIns + monthlyPmi;
   266→    const totalPaid = totalMonthly * n;
   267→    const totalInterest = pi * n - principal;
   268→
   269→    return {
   270→      principal,
   271→      pi,
   272→      monthlyPropTax,
   273→      monthlyIns,
   274→      monthlyPmi,
   275→      needsPmi,
   276→      downPct,
   277→      totalMonthly,
   278→      totalPaid,
   279→      totalInterest,
   280→      n,
   281→    };

   422→const COMPOUND_FREQ: Record<string, number> = {
   423→  annually: 1,
   424→  semiannually: 2,
   425→  quarterly: 4,
   426→  monthly: 12,
   427→  daily: 365,

   429→
   430→function CompoundInterestCalc() {
   431→  const [currency, setCurrency] = React.useState<string>("USD");
   432→  const [principal, setPrincipal] = React.useState("10000");
   433→  const [rate, setRate] = React.useState("7");
   434→  const [freq, setFreq] = React.useState("monthly");
   435→  const [years, setYears] = React.useState("10");
   436→  const [monthlyContribution, setMonthlyContribution] = React.useState("250");
   437→
   438→  const calc = React.useMemo(() => {
   439→    const P = parseNumber(principal);
   440→    const r = parseNumber(rate) / 100;
   441→    const t = parseNumber(years);
   442→    const n = COMPOUND_FREQ[freq] ?? 12;
   443→    const monthlyPmt = parseNumber(monthlyContribution);
   444→    // Convert monthly contribution to per-period contribution matching compounding frequency
   445→    const pmtPerPeriod = (monthlyPmt * 12) / n;
   446→
   447→    const fvPrincipal = fvLumpSum(P, r, n, t);
   448→    const fvContrib = fvAnnuity(pmtPerPeriod, r, n, t);
   449→    const amount = fvPrincipal + fvContrib;
   450→    const totalContributions = P + monthlyPmt * 12 * t;
   451→    const interestEarned = amount - totalContributions;
   452→
   453→    return { P, amount, totalContributions, interestEarned };
   454→  }, [principal, rate, freq, years, monthlyContribution]);
   455→
   456→  return (
   457→    <div className="space-y-6">
   458→      <div className="grid sm:grid-cols-2 gap-4">
   459→        <CurrencySelect value={currency} onChange={setCurrency} />
   460→        <div className="hidden sm:block" />
   461→        <Field label="Principal">
   462→          <Input value={principal} onChange={(e) => setPrincipal(e.target.value)} inputMode="decimal" />
   463→        </Field>
   464→        <Field label="Annual interest rate (%)">
   465→          <Input value={rate} onChange={(e) => setRate(e.target.value)} inputMode="decimal" />
   466→        </Field>
   467→        <Field label="Compounding frequency">
   468→          <Select value={freq} onValueChange={setFreq}>
   469→            <SelectTrigger className="w-full">
   470→              <SelectValue />
   471→            </SelectTrigger>
   472→            <SelectContent>
   473→              <SelectItem value="annually">Annually</SelectItem>
   474→              <SelectItem value="semiannually">Semi-annually</SelectItem>
   475→              <SelectItem value="quarterly">Quarterly</SelectItem>
   476→              <SelectItem value="monthly">Monthly</SelectItem>
   477→              <SelectItem value="daily">Daily</SelectItem>
   478→            </SelectContent>
   479→          </Select>
   480→        </Field>
   481→        <Field label="Time (years)">
   482→          <Input value={years} onChange={(e) => setYears(e.target.value)} inputMode="decimal" />
   483→        </Field>
   484→        <Field label="Monthly contribution (optional)">
   485→          <Input value={monthlyContribution} onChange={(e) => setMonthlyContribution(e.target.value)} inputMode="decimal" />
   486→        </Field>
   487→      </div>
   488→
   489→      <div className="grid sm:grid-cols-3 gap-4">
   490→        <CopyableStat label="Final Amount" displayValue={money(calc.amount, currency)} accent="primary" />
   491→        <ResultStat label="Total Contributions" value={money(calc.totalContributions, currency)} accent="default" />
   492→        <ResultStat label="Interest Earned" value={money(calc.interestEarned, currency)} accent="success" />
   493→      </div>
   494→
   495→      <FormulaBox>
   496→        A = P(1 + r/n)<sup>nt</sup> + PMT · [((1 + r/n)<sup>nt</sup> − 1) / (r/n)], where PMT is the contribution per
   497→        compounding period (monthly contribution × 12 ÷ n).
   498→      </FormulaBox>
   499→    </div>
   500→  );
   501→}

   507→function SimpleInterestCalc() {

   512→
   513→  const calc = React.useMemo(() => {

   866→/* 12. Discount                                                        */
   867→/* ------------------------------------------------------------------ */
   868→
   869→function DiscountCalc() {
   870→  const [currency, setCurrency] = React.useState<string>("USD");
   871→  const [mode, setMode] = React.useState<"forward" | "reverse">("forward");
   872→  const [original, setOriginal] = React.useState("120");
   873→  const [discount, setDiscount] = React.useState("20");
   874→  const [finalPrice, setFinalPrice] = React.useState("96");
   875→
   876→  const forward = React.useMemo(() => {
   877→    const orig = parseNumber(original);
   878→    const pct = parseNumber(discount);
   879→    const discountAmount = (orig * pct) / 100;
   880→    const final = orig - discountAmount;
   881→    return { orig, pct, discountAmount, final };
   882→  }, [original, discount]);
   883→
   884→  const reverse = React.useMemo(() => {
   885→    const final = parseNumber(finalPrice);
   886→    const pct = parseNumber(discount);
   887→    const orig = pct < 100 ? final / (1 - pct / 100) : final;
   888→    const discountAmount = orig - final;
   889→    return { orig, pct, discountAmount, final };
   890→  }, [finalPrice, discount]);
   891→
   892→  return (
   893→    <div className="space-y-6">
   894→      <div className="grid sm:grid-cols-2 gap-4">
   895→        <CurrencySelect value={currency} onChange={setCurrency} />
   896→        <Field label="Mode">
   897→          <Select value={mode} onValueChange={(v) => setMode(v as "forward" | "reverse")}>
   898→            <SelectTrigger className="w-full">
   899→              <SelectValue />
   900→            </SelectTrigger>
   901→            <SelectContent>
   902→              <SelectItem value="forward">Original → Final (apply discount)</SelectItem>
   903→              <SelectItem value="reverse">Final → Original (reverse discount)</SelectItem>
   904→            </SelectContent>
   905→          </Select>
   906→        </Field>
   907→      </div>
   908→
   909→      {mode === "forward" ? (
   910→        <div className="grid sm:grid-cols-2 gap-4">
   911→          <Field label="Original price">
   912→            <Input value={original} onChange={(e) => setOriginal(e.target.value)} inputMode="decimal" />
   913→          </Field>
   914→          <Field label="Discount (%)">
   915→            <Input value={discount} onChange={(e) => setDiscount(e.target.value)} inputMode="decimal" />
   916→          </Field>
   917→        </div>
   918→      ) : (
   919→        <div className="grid sm:grid-cols-2 gap-4">
   920→          <Field label="Final price (after discount)">
   921→            <Input value={finalPrice} onChange={(e) => setFinalPrice(e.target.value)} inputMode="decimal" />
   922→          </Field>
   923→          <Field label="Discount (%)">
   924→            <Input value={discount} onChange={(e) => setDiscount(e.target.value)} inputMode="decimal" />
   925→          </Field>
   926→        </div>
   927→      )}
   928→
   929→      {mode === "forward" ? (
   930→        <div className="grid sm:grid-cols-3 gap-4">
   931→          <CopyableStat label="Final Price" displayValue={money(forward.final, currency)} accent="primary" />
   932→          <ResultStat label="You Save" value={money(forward.discountAmount, currency)} accent="success" />
   933→          <ResultStat label="Discount Amount" value={money(forward.discountAmount, currency)} accent="default" />
   934→        </div>
   935→      ) : (
   936→        <div className="grid sm:grid-cols-3 gap-4">
   937→          <CopyableStat label="Original Price" displayValue={money(reverse.orig, currency)} accent="primary" />
   938→          <ResultStat label="Discount Amount" value={money(reverse.discountAmount, currency)} accent="warning" />
   939→          <ResultStat label="Final Price" value={money(reverse.final, currency)} accent="default" />
   940→        </div>
   941→      )}
   942→
   943→      <FormulaBox>
   944→        Forward: Final = Original × (1 − Discount/100). Reverse: Original = Final ÷ (1 − Discount/100). Savings =
   945→        Original − Final.
   946→      </FormulaBox>
   947→    </div>
   948→  );
   949→}

  1061→function BreakEvenCalc() {
  1062→  const [currency, setCurrency] = React.useState<string>("USD");
  1063→  const [fixed, setFixed] = React.useState("50000");
  1064→  const [pricePerUnit, setPricePerUnit] = React.useState("25");

  1068→    const f = parseNumber(fixed);
  1069→    const p = parseNumber(pricePerUnit);
  1070→    const v = parseNumber(variablePerUnit);
  1071→    const contribution = p - v;
  1072→    const units = contribution > 0 ? f / contribution : 0;
  1073→    const revenue = units * p;
  1074→    const contributionMargin = p > 0 ? (contribution / p) * 100 : 0;
  1075→    return { f, p, v, contribution, units, revenue, contributionMargin };
  1076→  }, [fixed, pricePerUnit, variablePerUnit]);
  1077→
  1078→  return (
  1079→    <div className="space-y-6">
  1080→      <div className="grid sm:grid-cols-2 gap-4">
  1081→        <CurrencySelect value={currency} onChange={setCurrency} />
  1082→        <div className="hidden sm:block" />
  1083→        <Field label="Fixed costs">
  1084→          <Input value={fixed} onChange={(e) => setFixed(e.target.value)} inputMode="decimal" />
  1085→        </Field>
  1086→        <Field label="Price per unit">
  1087→          <Input value={pricePerUnit} onChange={(e) => setPricePerUnit(e.target.value)} inputMode="decimal" />
  1088→        </Field>
  1089→        <Field label="Variable cost per unit">
  1090→          <Input value={variablePerUnit} onChange={(e) => setVariablePerUnit(e.target.value)} inputMode="decimal" />
  1091→        </Field>
  1092→        <Field label="Contribution margin per unit">
  1093→          <Input value={money(calc.contribution, currency)} readOnly className="bg-muted/40 font-mono" />
  1094→        </Field>
  1095→      </div>
  1096→
  1097→      <div className="grid sm:grid-cols-3 gap-4">
  1098→        <CopyableStat label="Break-Even Units" displayValue={num(calc.units, 0)} accent="primary" sublabel={`Revenue: ${money(calc.revenue, currency)}`} />
  1099→        <ResultStat label="Break-Even Revenue" value={money(calc.revenue, currency)} accent="default" />
  1100→        <ResultStat label="Contribution Margin" value={`${num(calc.contributionMargin, 1)}%`} accent="default" />
  1101→      </div>
  1102→
  1103→      {calc.contribution <= 0 && (
  1104→        <div className="rounded-xl border border-warning/40 bg-warning/5 p-4 text-sm text-foreground">
  1105→          Price per unit must be greater than variable cost per unit to reach break-even. Currently your contribution
  1106→          margin per unit is {money(calc.contribution, currency)}.
  1107→        </div>
  1108→      )}
  1109→
  1110→      <FormulaBox>
  1111→        Break-even units = Fixed costs ÷ (Price per unit − Variable cost per unit). Break-even revenue = Break-even
  1112→        units × Price per unit.
  1113→      </FormulaBox>
  1114→    </div>
  1115→  );
  1116→}

  1210→function RetirementCalc() {
  1211→  const [currency, setCurrency] = React.useState<string>("USD");
  1212→  const [currentAge, setCurrentAge] = React.useState("30");
  1213→  const [retireAge, setRetireAge] = React.useState("65");
  1214→  const [currentSavings, setCurrentSavings] = React.useState("25000");
  1215→  const [monthly, setMonthly] = React.useState("500");
  1216→  const [rate, setRate] = React.useState("7");
  1217→
  1218→  const calc = React.useMemo(() => {
  1219→    const ageNow = parseNumber(currentAge);
  1220→    const ageRetire = parseNumber(retireAge);
  1221→    const years = Math.max(0, ageRetire - ageNow);
  1222→    const months = years * 12;
  1223→    const P = parseNumber(currentSavings);
  1224→    const pmtVal = parseNumber(monthly);
  1225→    const r = parseNumber(rate) / 100;
  1226→
  1227→    const fvPrincipal = fvLumpSum(P, r, 12, years);
  1228→    const fvContrib = fvAnnuity(pmtVal, r, 12, years);
  1229→    const corpus = fvPrincipal + fvContrib;
  1230→    const totalContributions = P + pmtVal * months;
  1231→    const growth = corpus - totalContributions;
  1232→    return { years, months, corpus, totalContributions, growth };
  1233→  }, [currentAge, retireAge, currentSavings, monthly, rate]);
  1234→
  1235→  return (
  1236→    <div className="space-y-6">
  1237→      <div className="grid sm:grid-cols-2 gap-4">
  1238→        <CurrencySelect value={currency} onChange={setCurrency} />
  1239→        <div className="hidden sm:block" />
  1240→        <Field label="Current age">
  1241→          <Input value={currentAge} onChange={(e) => setCurrentAge(e.target.value)} inputMode="decimal" />
  1242→        </Field>
  1243→        <Field label="Retirement age">
  1244→          <Input value={retireAge} onChange={(e) => setRetireAge(e.target.value)} inputMode="decimal" />
  1245→        </Field>
  1246→        <Field label="Current savings">
  1247→          <Input value={currentSavings} onChange={(e) => setCurrentSavings(e.target.value)} inputMode="decimal" />
  1248→        </Field>
  1249→        <Field label="Monthly contribution">
  1250→          <Input value={monthly} onChange={(e) => setMonthly(e.target.value)} inputMode="decimal" />
  1251→        </Field>
  1252→        <Field label="Expected annual return (%)">
  1253→          <Input value={rate} onChange={(e) => setRate(e.target.value)} inputMode="decimal" />
  1254→        </Field>
  1255→        <div className="text-xs text-muted-foreground self-end pb-2">
  1256→          {calc.years} years · {num(calc.months, 0)} monthly contributions
  1257→        </div>
  1258→      </div>
  1259→
  1260→      <div className="grid sm:grid-cols-3 gap-4">
  1261→        <CopyableStat
  1262→          label="Projected Corpus"
  1263→          displayValue={money(calc.corpus, currency)}
  1264→          accent="primary"
  1265→          sublabel={`at retirement`}
  1266→        />
  1267→        <ResultStat label="Total Contributions" value={money(calc.totalContributions, currency)} accent="default" />
  1268→        <ResultStat label="Growth" value={money(calc.growth, currency)} accent="success" />
  1269→      </div>
  1270→
  1271→      <FormulaBox>
  1272→        FV = P(1 + r/12)<sup>12t</sup> + PMT · [((1 + r/12)<sup>12t</sup> − 1) / (r/12)], where t = retirement age −
  1273→        current age.
  1274→      </FormulaBox>
  1275→    </div>
  1276→  );
  1277→}
